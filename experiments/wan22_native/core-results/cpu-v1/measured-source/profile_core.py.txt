"""Guarded single native5B positive/negative pair, not a video generation loop.

Requires completed new48-channel observation-only codec evidence, genuine text,
and source-bound CPU checks. The parent enforces900s/18GiB/minimum2GiB memory.
The worker separately samples Metal allocation during loading and inference.
"""
import argparse
import ast
import importlib.metadata
import json
import os
from pathlib import Path
import shutil
import subprocess
import sys
import threading
import time

import psutil
from safetensors.torch import load_file, save_file
import torch

from experiments.room_world.memory_train import atomic_write, handoff_and_guard, new_directory
from experiments.wan_adapter.text_cache.cache import load_context
from experiments.wan_adapter.native_control.vendor.fm_solvers_unipc import FlowUniPCMultistepScheduler
from .load_weights import load_core, sha256, tensor_sha256
from .portable import NATIVE_GRID, NATIVE_LATENT_SHAPE, NATIVE_TOKENS, token_times

HERE = Path(__file__).parent
REPO = HERE.parent.parent
SOURCE_NAMES = ("portable.py", "load_weights.py", "profile_core.py", "test_core.py", "config.json", "weights.index.json",
                "provenance.json", "text-reuse.json", "vendor/model.py", "vendor/attention.py", "vendor/shared_config.py.txt")
REUSED_NAMES = ("experiments/wan_adapter/native_control/portable.py",
                "experiments/wan_adapter/native_control/vendor/fm_solvers_unipc.py",
                "experiments/wan_adapter/text_cache/cache.py", "experiments/room_world/memory_train.py")
STEPS, SHIFT, GUIDANCE, SEED = 50, 5., 5., 20260908
IMAGE_SHA256 = "7bdfa121eb2917b837af3ee1faae9e697751cdd3cdee0b21422c1b1f2a53e780"


def make_scheduler():
    scheduler = FlowUniPCMultistepScheduler(num_train_timesteps=1000, shift=1, use_dynamic_shifting=False)
    scheduler.set_timesteps(STEPS, device="cpu", shift=SHIFT)
    return scheduler


def prepare_input(observation, timestep, seed=SEED):
    if (tuple(observation.shape) != (1, 48, 1, 18, 32) or observation.dtype != torch.float32
            or observation.device.type != "cpu" or not torch.isfinite(observation).all()):
        raise ValueError("Only the new native48-channel FP32 initial observation cache is accepted")
    if type(seed) is not int or seed < 0 or seed >= 2**63:
        raise ValueError("Seed must be an integer in[0,2**63)")
    noise = torch.randn(NATIVE_LATENT_SHAPE, generator=torch.Generator(device="cpu").manual_seed(seed), dtype=torch.float32)
    latent = noise.clone()
    latent[:, :1] = observation[0]
    times = token_times(torch.tensor([NATIVE_GRID]), torch.as_tensor(timestep, dtype=torch.int64).reshape(1), NATIVE_TOKENS)
    return noise, latent, times


def load_observation(directory):
    root = Path(directory).resolve()
    report_path = root / "metrics.json"
    report = json.loads(report_path.read_text())
    if (report.get("status") != "passed" or report.get("finite_output") is not True
            or report.get("input_rgb_frames") != 1 or report.get("input_image_sha256") != IMAGE_SHA256
            or report.get("future_rgb_read") is not False or report.get("old_latent_read") is not False
            or report.get("actions_read") is not False or report.get("target_read") is not False
            or report.get("expected_latent_shape") != [1, 48, 1, 18, 32]
            or report.get("cache_clear_after_encode") is not True or report.get("cache_clear_after_decode") is not True):
        raise ValueError("A passed new-VAE observation-only codec profile is required")
    expected_sources = {"codec.py", "codec_profile.py", "codec-source.json", "test_codec.py", "vendor/vae2_2.py"}
    provenance = json.loads((HERE / "provenance.json").read_text())
    if (not expected_sources.issubset(report.get("source_sha256", {}))
            or report.get("codec", {}).get("weight_sha256") != provenance["native_vae_sha256"]
            or sha256(root / "original.png") != IMAGE_SHA256):
        raise ValueError("Codec source, native weight or actual original image identity differs")
    for name, expected in report["source_sha256"].items():
        path = (HERE / name).resolve()
        if not path.is_relative_to(HERE.resolve()) or not path.is_file() or sha256(path) != expected:
            raise ValueError("Native codec source differs from its executed cache profile")
    cache = root / "observation.safetensors"
    if not cache.resolve().is_relative_to(root) or cache.stat().st_size > 2**20 or sha256(cache) != report["output_sha256"][cache.name]:
        raise ValueError("Native observation cache file differs")
    values = load_file(str(cache), device="cpu")
    if set(values) != {"observation"} or tensor_sha256(values["observation"]) != report["latent_tensor_sha256"]:
        raise ValueError("Native observation tensor identity differs")
    prepare_input(values["observation"], torch.tensor(999))
    return values["observation"], {"codec_report_sha256": sha256(report_path), "cache_sha256": sha256(cache),
        "observation_tensor_sha256": report["latent_tensor_sha256"], "input_image_sha256": IMAGE_SHA256,
        "codec_source_sha256": report["source_sha256"]}


def load_text(directory):
    reuse = json.loads((HERE / "text-reuse.json").read_text())
    manifest_path = Path(directory) / "manifest.json"
    if reuse["status"] != "verified" or sha256(manifest_path) != reuse["text_cache_manifest_sha256"]:
        raise ValueError("The exact verified genuine UMT5 cache manifest is required")
    source = HERE / reuse["negative_prompt_source"]["file"]
    if sha256(source) != reuse["negative_prompt_source"]["sha256"]:
        raise ValueError("Pinned native negative-prompt source changed")
    negative = None
    for node in ast.parse(source.read_text()).body:
        if isinstance(node, ast.Assign) and any(isinstance(t, ast.Attribute) and t.attr == "sample_neg_prompt" for t in node.targets):
            negative = ast.literal_eval(node.value)
    if not isinstance(negative, str) or not negative:
        raise ValueError("Pinned configured negative prompt is missing")
    positive = load_context(directory, "atrium", expected_text=reuse["positive_text"], dtype=torch.float32)
    negative_context = load_context(directory, "native_negative", expected_text=negative, dtype=torch.float32)
    return positive, negative_context, {"manifest_sha256": sha256(manifest_path), "reuse_evidence_sha256": sha256(HERE / "text-reuse.json"),
        "positive_tensor_sha256": tensor_sha256(positive), "negative_tensor_sha256": tensor_sha256(negative_context)}


def validate_cpu(path, *, independent=False):
    report = json.loads(Path(path).read_text())
    if report.get("status") != "passed" or report.get("tests", report.get("tests_run", 0)) < (4 if independent else 12):
        raise ValueError("Passed source-bound CPU implementation and independent checks are required")
    required = ("portable.py", "load_weights.py", "vendor/model.py", "vendor/attention.py") if independent else SOURCE_NAMES
    for name in required:
        hashes = report.get("source_sha256", {})
        if hashes.get(name, hashes.get("experiments/wan22_native/" + name)) != sha256(HERE / name):
            raise ValueError("Core source changed after CPU verification: " + name)
    if not independent:
        for name in REUSED_NAMES:
            if report.get("reused_source_sha256", {}).get(name) != sha256(REPO / name):
                raise ValueError("Imported helper source changed after CPU verification: " + name)
    return sha256(path)


def worker(config):
    out = Path(config["output"])
    if set(config["source_sha256"]) != set(SOURCE_NAMES) or set(config["reused_source_sha256"]) != set(REUSED_NAMES):
        raise ValueError("Incomplete core source snapshot")
    for name, expected in config["source_sha256"].items():
        if sha256(HERE / name) != expected:
            raise ValueError("Core source changed after the parent snapshot")
    for name, expected in config["reused_source_sha256"].items():
        if sha256(REPO / name) != expected:
            raise ValueError("Imported helper source changed after the parent snapshot")
    device = config["device"]
    if device == "mps" and not torch.backends.mps.is_available():
        raise RuntimeError("MPS unavailable")
    if os.environ.get("PYTORCH_ENABLE_MPS_FALLBACK", "0") != "0":
        raise RuntimeError("Automatic CPU fallback must be disabled")
    torch.set_num_threads(2)
    if device == "mps":
        torch.mps.set_per_process_memory_fraction(min(1., 18 * 2**30 / torch.mps.recommended_max_memory()))
    started, stop = time.monotonic(), threading.Event()
    peaks = {"rss_bytes": 0, "mps_active_bytes": 0, "mps_driver_bytes": 0}
    report = {"status": "running", "external_model": "Official Wan2.2 TI2V-5B with explicit selectiveBF16 port",
        "original_worldline_model": False, "full_video_generated": False, "quality_measured": False,
        "device": device, "seed": SEED, "sampling_configuration": {"steps": STEPS, "shift": SHIFT, "guidance": GUIDANCE},
        "latent_shape": list(NATIVE_LATENT_SHAPE), "tokens": 720, "observed_tokens": 144,
        "timings": [], "source_sha256": config["source_sha256"], "reused_source_sha256": config["reused_source_sha256"], "max_seconds": 900,
        "max_memory_gib": 18, "minimum_available_gib": 2, "stage": "input validation",
        "dependencies": {name: importlib.metadata.version(name) for name in ("torch", "diffusers", "numpy", "safetensors", "psutil")}}

    def save():
        report["peaks_sampled"] = dict(peaks)
        atomic_write(out / "metrics.json", report)

    def check():
        rss = psutil.Process().memory_info().rss
        available = psutil.virtual_memory().available
        driver = torch.mps.driver_allocated_memory() if device == "mps" else 0
        if max(rss, driver) > 18 * 2**30 or available < 2 * 2**30 or time.monotonic() - started > 900:
            raise RuntimeError("Bounded core loading/inference resource limit reached")

    def monitor():
        with (out / "memory.jsonl").open("x") as stream:
            while not stop.is_set():
                row = {"seconds": time.monotonic() - started, "stage": report["stage"],
                    "rss_bytes": psutil.Process().memory_info().rss, "available_bytes": psutil.virtual_memory().available,
                    "mps_active_bytes": 0, "mps_driver_bytes": 0}
                if device == "mps":
                    row.update(mps_active_bytes=torch.mps.current_allocated_memory(), mps_driver_bytes=torch.mps.driver_allocated_memory())
                for key in peaks:
                    peaks[key] = max(peaks[key], row[key])
                stream.write(json.dumps(row) + "\n")
                stream.flush()
                if max(row["rss_bytes"], row["mps_driver_bytes"]) > 18 * 2**30 or row["available_bytes"] < 2 * 2**30 or row["seconds"] > 900:
                    atomic_write(out / "watchdog-stop.json", {"status": "stopped", "sample": row})
                    os._exit(124)
                stop.wait(.5)

    def measured(stage, call):
        report["stage"] = stage
        check()
        if device == "mps":
            torch.mps.synchronize()
        tick = time.monotonic()
        value = call()
        if device == "mps":
            torch.mps.synchronize()
        report["timings"].append({"stage": stage, "seconds": time.monotonic() - tick})
        check()
        save()
        return value

    save()
    watcher = threading.Thread(target=monitor, daemon=True)
    watcher.start()
    try:
        observation, report["observation"] = measured("verify new observation-only cache", lambda: load_observation(config["observation_cache"]))
        positive, negative, report["text"] = measured("verify genuine text", lambda: load_text(config["text_cache"]))
        scheduler = make_scheduler()
        time0 = scheduler.timesteps[0]
        noise, latent_cpu, times_cpu = prepare_input(observation, time0)
        save_file({"initial_noise": noise, "initial_latent": latent_cpu, "observation": observation, "token_times": times_cpu}, str(out / "inputs.safetensors"))
        model, load_report = measured("verify and stream official transformer", lambda: load_core(config["weights"], device=device, check=check))
        atomic_write(out / "weight-load.json", load_report)
        report["weight_load_sha256"] = sha256(out / "weight-load.json")
        latent, times = latent_cpu.to(device), times_cpu.to(device)
        positive, negative = positive.to(device), negative.to(device)
        with torch.inference_mode():
            v_positive = measured("positive native forward", lambda: model([latent], times, [positive], 720)[0])
            v_negative = measured("negative native forward", lambda: model([latent], times, [negative], 720)[0])
            if any(v.dtype != torch.float32 or tuple(v.shape) != NATIVE_LATENT_SHAPE or not torch.isfinite(v).all() for v in (v_positive, v_negative)):
                raise FloatingPointError("Native forward must return finiteFP32 velocity at the expected shape")
            guided = v_negative + GUIDANCE * (v_positive - v_negative)
            if not torch.isfinite(guided).all():
                raise FloatingPointError("FP32 guidance is nonfinite")
            velocity_cpu = measured("guided velocity transfer to CPU solver", lambda: guided.cpu())
            next_cpu = measured("one official CPU UniPC step", lambda: scheduler.step(velocity_cpu.unsqueeze(0), time0,
                latent_cpu.unsqueeze(0), return_dict=False)[0].squeeze(0))
            next_cpu[:, :1] = observation[0]
            if next_cpu.dtype != torch.float32 or not torch.isfinite(next_cpu).all() or not torch.equal(next_cpu[:, :1], observation[0]):
                raise FloatingPointError("CPU solver/prefix restoration failed")
            save_file({"positive_velocity": v_positive.cpu().contiguous(), "negative_velocity": v_negative.cpu().contiguous(),
                "guided_velocity": velocity_cpu.contiguous(), "one_step_latent": next_cpu.contiguous()}, str(out / "outputs.safetensors"))
        report.update(status="passed", finite_outputs=True, observed_prefix_exact=True,
            output_sha256={name: sha256(out / name) for name in ("inputs.safetensors", "outputs.safetensors", "weight-load.json")})
        report["pair_seconds"] = sum(r["seconds"] for r in report["timings"] if r["stage"] in ("positive native forward", "negative native forward"))
        report["fifty_pairs_linear_estimate_seconds"] = 50 * report["pair_seconds"]
        report["estimate_limit"] = "One pair only; excludes decode, nonlinear warmup effects and later-step variation. No generation-budget or image-quality gate is passed automatically."
    except BaseException as error:
        report.update(status="interrupted" if isinstance(error, KeyboardInterrupt) else "failed", error_type=type(error).__name__, error=str(error))
        raise
    finally:
        stop.set()
        watcher.join(timeout=2)
        report["elapsed_seconds"] = time.monotonic() - started
        save()


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--worker", action="store_true", help=argparse.SUPPRESS)
    for name in ("weights", "observation-cache", "text-cache", "cpu-report", "independent-report", "output"):
        parser.add_argument("--" + name, type=Path)
    parser.add_argument("--device", choices=("cpu", "mps"), default="cpu")
    args = parser.parse_args()
    if args.worker:
        worker(json.loads(sys.stdin.read(65536)))
        return
    if any(getattr(args, name) is None for name in ("weights", "observation_cache", "text_cache", "cpu_report", "independent_report", "output")):
        parser.error("All input artifacts, CPU review reports and a new output directory are required")
    if args.output.resolve().is_relative_to(HERE.resolve()):
        raise ValueError("Actual profile output must be outside the source package")
    evidence = {"cpu_report_sha256": validate_cpu(args.cpu_report),
                "independent_report_sha256": validate_cpu(args.independent_report, independent=True)}
    # Input checks precede creating the worker or loading any real model parameter.
    load_observation(args.observation_cache)
    load_text(args.text_cache)
    out = new_directory(args.output)
    sources = {name: sha256(HERE / name) for name in SOURCE_NAMES}
    reused = {name: sha256(REPO / name) for name in REUSED_NAMES}
    for name in SOURCE_NAMES:
        target = out / "measured-source" / (name + ".txt")
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copyfile(HERE / name, target)
        if sha256(target) != sources[name]:
            raise RuntimeError("Source changed while snapshotting")
    for name, expected in reused.items():
        target = out / "reused-source" / (name + ".txt")
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copyfile(REPO / name, target)
        if sha256(target) != expected:
            raise RuntimeError("Imported source changed while snapshotting")
    config = {name: str(getattr(args, name).resolve()) for name in ("weights", "observation_cache", "text_cache", "output")}
    config.update(device=args.device, source_sha256=sources, reused_source_sha256=reused)
    atomic_write(out / "launch.json", dict(config, **evidence))
    with (out / "worker.log").open("x") as log:
        process = subprocess.Popen([sys.executable, "-m", "experiments.wan22_native.profile_core", "--worker"],
            stdin=subprocess.PIPE, stdout=log, stderr=subprocess.STDOUT, env=dict(os.environ, PYTORCH_ENABLE_MPS_FALLBACK="0"))
        terminal = handoff_and_guard(process, config, out, max_seconds=900, max_rss_gib=18, minimum_available_gib=2)
    print(json.dumps({"status": terminal["status"], **evidence}))
    if terminal["status"] != "complete":
        raise SystemExit(1)


if __name__ == "__main__":
    main()
