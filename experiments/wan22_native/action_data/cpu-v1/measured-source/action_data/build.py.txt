"""Plan the eight-window cache, or explicitly launch its guarded codec process."""
import argparse
from pathlib import Path
import json

from ..codec_profile import atomic_json
from . import data
from .runtime import gates, new_output, snapshot, run_child, source_hashes


def validate_roundtrip(directory):
    directory=Path(directory);r=json.loads((directory/'result/metrics.json').read_text())
    terminal=json.loads((directory/'terminal.json').read_text())
    if (terminal.get('status')!='complete' or terminal.get('exit_code')!=0 or r.get('status')!='passed'
            or r.get('mode')!='roundtrip' or not r.get('finite_output')
            or (directory/'result/watchdog-stop.json').exists() or not r.get('cache_clear_after_operations')
            or r.get('source_sha256')!=source_hashes()
            or len(r.get('reconstruction',{}).get('per_frame',[]))!=17
            or not all(c.get('passed') is True for c in r.get('causal_checks',[])) or not r.get('causal_checks')):
        raise ValueError('Completed source-matching real 17-frame reconstruction required before cache encoding')
    return data.sha(directory/'result/metrics.json')


def run(mode):
    parser=argparse.ArgumentParser(description=__doc__)
    for name in ('capture','output'):parser.add_argument('--'+name,type=Path,required=True)
    parser.add_argument('--execute',action='store_true',help='Explicitly run one bounded codec child; default only validates original input data')
    parser.add_argument('--weights',type=Path)
    parser.add_argument('--device',choices=('cpu','mps'),default='mps')
    parser.add_argument('--cpu-report',type=Path,default=data.HERE/'cpu-v1/tests.json')
    parser.add_argument('--decoder-run',type=Path,default=data.HERE.parent/'codec-results/full-decode-v3')
    parser.add_argument('--roundtrip-run',type=Path)
    args=parser.parse_args()
    if args.execute and args.weights is None:parser.error('--execute requires --weights')
    if args.execute and mode=='cache' and args.roundtrip_run is None:parser.error('Cache execution requires --roundtrip-run')
    output=new_output(args.output)
    if mode=='cache':plan=data.plan(args.capture)
    else:
        rgb,provenance=data.load_roundtrip_rgb(args.capture)
        plan={'schema':'worldline-wan22-atrium-roundtrip-plan-v1','status':'planned','mode':'roundtrip',
            'source':provenance,'rgb_shape':list(rgb.shape),'target_shape':[1,48,5,18,32],
            'observation_shape':[1,48,1,18,32],'decode_shape':[1,3,17,288,512],
            'rgb_frames':17,'action_values_materialized':False,'model_execution':False,'gpu_execution':False,
            'raw_capture_modified':False,'metrics':'All-frame, first-frame, later-frame and each-frame FP32 MSE/MAE/PSNR',
            'scope':'Codec reconstruction only, without a transformer, actions, sampling or new generated futures'}
    plan['source_sha256']=snapshot(output);atomic_json(output/'plan.json',plan)
    if args.execute:
        validations=gates(args.cpu_report,args.decoder_run)
        if mode=='cache':validations['roundtrip_metrics_sha256']=validate_roundtrip(args.roundtrip_run)
        atomic_json(output/'gates.json',validations)
        run_child(output,{'mode':mode,'capture':str(args.capture.resolve()),'device':args.device,
            'weights':str(args.weights.resolve()),'cpu_report':str(args.cpu_report.resolve()),
            'decoder_run':str(args.decoder_run.resolve()),'plan_sha256':data.sha(output/'plan.json')})
    print(json.dumps({'status':'complete' if args.execute else 'planned','mode':mode,'model_execution':args.execute,
        'plan_sha256':data.sha(output/'plan.json')},indent=2))


def main():run('cache')
if __name__=='__main__':main()
