def sha(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as handle:
        for block in iter(lambda: handle.read(2**20), b''):
            h.update(block)
    return h.hexdigest()


def read(path):
    return json.loads(Path(path).read_text())


def tensor_hashes(values):
    return {name: {'shape': list(value.shape), 'dtype': str(value.dtype),
        'sha256': hashlib.sha256(value.detach().cpu().contiguous().view(torch.uint8).numpy().tobytes()).hexdigest()}
        for name, value in sorted(values.items())}


def finite(value):
    if isinstance(value, torch.Tensor):
        assert torch.isfinite(value).all()
    elif isinstance(value, dict):
        for item in value.values(): finite(item)
    elif isinstance(value, (tuple, list)):
        for item in value: finite(item)
    elif isinstance(value, float):
        assert math.isfinite(value)


def file_inventory(root):
    return {str(path.relative_to(root)): {'bytes': path.stat().st_size, 'sha256': sha(path)}
            for path in sorted(root.rglob('*')) if path.is_file()}

def expect(actual, expected):
    if isinstance(expected, dict):
        for key, value in expected.items(): expect(actual[key],value)
    elif isinstance(expected,(list,tuple)):
        assert len(actual)==len(expected)
        for a,b in zip(actual,expected):expect(a,b)
    elif expected is None or isinstance(expected,(str,bool)):
        assert actual==expected,(actual,expected)
    else:
        assert math.isfinite(actual) and math.isfinite(expected)
        delta=abs(float(actual)-float(expected));NUMERIC_CHECKS['count']+=1
        NUMERIC_CHECKS['maximum_absolute_difference']=max(NUMERIC_CHECKS['maximum_absolute_difference'],delta)
        assert math.isclose(actual,expected,rel_tol=1e-9,abs_tol=1e-12),(actual,expected)


def interval(values):
    values=np.asarray(values,dtype=np.float64);valid=np.isfinite(values)
    return {'percentile_95':np.percentile(values,[2.5,97.5]).tolist() if valid.all() else None,
            'defined_resamples':int(valid.sum()),'undefined_resamples':int((~valid).sum())}
