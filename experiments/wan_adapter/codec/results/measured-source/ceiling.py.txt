"""Measure official Wan VAE reconstruction and first-frame causal consistency.

This is an external pretrained codec reconstruction test on original imagery,
not a generated-video quality test. No model training or text encoder is used.
"""
import argparse
import hashlib
import json
import math
import os
from pathlib import Path
import shutil
import sys
import threading
import time
import platform
sys.path.insert(0,str(Path(__file__).resolve().parent.parent))
import numpy as np
from PIL import Image, ImageDraw
import psutil
from safetensors.torch import save_file
from skimage.metrics import structural_similarity
import torch
from codec.helper import OfficialWanCodec, VAE_SHA256, sha

GiB=1024**3

def save_json(path,data):
    temp=path.with_suffix('.tmp')
    temp.write_text(json.dumps(data,indent=2)+'\n');temp.replace(path)


def main():
    p=argparse.ArgumentParser(description=__doc__)
    p.add_argument('--weights',type=Path,required=True)
    p.add_argument('--frames-dir',type=Path,required=True)
    p.add_argument('--output',type=Path,required=True)
    p.add_argument('--start',type=int,default=0)
    p.add_argument('--frames',type=int,default=17)
    p.add_argument('--device',choices=['cpu','mps'],default='mps')
    p.add_argument('--dtype',choices=['float32','float16'],default='float32')
    a=p.parse_args()
    if a.frames != 17:
        p.error('This bounded reconstruction test uses exactly 17 frames')
    a.output.mkdir(parents=True,exist_ok=True)
    for d in ['original','reconstructed']:
        (a.output/d).mkdir(exist_ok=True)
    torch.set_num_threads(4)
    if a.device=='mps':
        if not torch.backends.mps.is_available():raise RuntimeError('MPS unavailable')
        torch.mps.set_per_process_memory_fraction(min(1.,18*GiB/torch.mps.recommended_max_memory()))
    files=[a.frames_dir/f'{i:04d}.png' for i in range(a.start,a.start+a.frames)]
    images=[np.asarray(Image.open(f).convert('RGB'),dtype=np.uint8) for f in files]
    if any(x.shape!=(288,512,3) for x in images):raise ValueError('Expected original native 512x288 RGB frames, no resizing')
    source=np.stack(images).astype(np.float32)/255.
    video=torch.from_numpy(source).permute(3,0,1,2).unsqueeze(0)*2-1
    report={'status':'running','experiment':'official external Wan VAE reconstruction on original Blender capture',
        'generated_video_quality_evidence':False,'device':a.device,'dtype':a.dtype,
        'torch':torch.__version__,'platform':platform.platform(),
        'automatic_mps_cpu_fallback':os.environ.get('PYTORCH_ENABLE_MPS_FALLBACK','0'),
        'mps_allocator_cache_policy':'empty unused allocator cache after each load/encode/decode stage; timing includes cleanup',
        'weights_sha256':VAE_SHA256,'source_rgb_transform':'original PNG sRGB with Blender AgX view; no crop/resize',
        'pixel_shape':[1,3,17,288,512],'source_arm':a.frames_dir.name,'source_start':a.start,
        'input_images':[{'file':f.name,'sha256':sha(f)} for f in files],
        'input_license':'CC0-1.0 original procedural Blender scene',
        'psnr_definition':'RGB float32 [0,1], whole-clip MSE, 10*log10(1/MSE)',
        'ssim_definition':'skimage structural_similarity RGB mean, Gaussian sigma1.5, 11-pixel window, population covariance, data_range1',
        'causal_check':'same first RGB frame encoded alone versus first latent frame of all17 RGB frames',
        'causal_tolerance':{'atol':1e-5,'rtol':1e-5}}
    started=time.perf_counter();stage=['load'];stop=threading.Event();peak={'rss_bytes':0,'mps_active_bytes':0,'mps_driver_bytes':0};proc=psutil.Process()
    save_json(a.output/'metrics.json',report)
    def monitor():
        with (a.output/'memory.jsonl').open('w') as log:
            while not stop.is_set():
                row={'seconds':time.perf_counter()-started,'stage':stage[0],'rss_bytes':proc.memory_info().rss,'available_system_bytes':psutil.virtual_memory().available}
                if a.device=='mps':row.update(mps_active_bytes=torch.mps.current_allocated_memory(),mps_driver_bytes=torch.mps.driver_allocated_memory())
                for key in peak:peak[key]=max(peak[key],row.get(key,0))
                log.write(json.dumps(row)+'\n');log.flush()
                reason=None
                if row['seconds']>600:reason='time-cap-600-seconds'
                elif max(row['rss_bytes'],row.get('mps_driver_bytes',0))>18*GiB:reason='memory-cap-18-GiB'
                elif row['available_system_bytes']<2*GiB:reason='system-available-below-2-GiB'
                if reason:
                    save_json(a.output/'watchdog-stop.json',{'reason':reason,'last_sample':row});os._exit(124)
                stop.wait(.5)
    thread=threading.Thread(target=monitor,daemon=True);thread.start()
    def sync():
        if a.device=='mps':torch.mps.synchronize()
    def timed(label,func):
        stage[0]=label;sync();t=time.perf_counter();out=func();sync()
        if a.device=='mps':torch.mps.empty_cache()
        report[label+'_seconds']=time.perf_counter()-t
        save_json(a.output/'metrics.json',report)
        return out
    try:
        codec=timed('load',lambda:OfficialWanCodec(a.weights,a.device,getattr(torch,a.dtype)))
        report['vae_parameters']=sum(p.numel() for p in codec.model.parameters())
        report['loaded_keys']=codec.loaded_keys;report['missing_keys']=codec.missing_keys;report['unexpected_keys']=codec.unexpected_keys
        first=timed('first_frame_encode',lambda:codec.encode(video[:,:,:1]))
        latent=timed('full_clip_encode',lambda:codec.encode(video))
        report['latent_shape']=list(latent.shape)
        if list(latent.shape)!=[1,16,5,36,64]:raise RuntimeError('Unexpected latent shape')
        if not torch.isfinite(latent).all().item() or not torch.isfinite(first).all().item():raise RuntimeError('Non-finite encoded latents')
        difference=(first-latent[:,:,:1]).abs()
        report['first_latent_max_abs_difference']=difference.max().item()
        report['first_latent_mean_abs_difference']=difference.mean().item()
        report['first_latent_causal_check_passed']=torch.allclose(first,latent[:,:,:1],atol=1e-5,rtol=1e-5)
        save_json(a.output/'metrics.json',report)
        if not report['first_latent_causal_check_passed']:raise RuntimeError('First-latent causal consistency check failed')
        reconstruction=timed('full_clip_decode',lambda:codec.decode(latent))
        if not torch.isfinite(reconstruction).all().item():raise RuntimeError('Non-finite reconstructed pixels')
        report['all_latents_and_reconstructions_finite']=True
        reconstructed=((reconstruction[0].permute(1,2,3,0).cpu().numpy()+1)/2).clip(0,1)
        mse=float(np.mean((source.astype(np.float64)-reconstructed)**2))
        report['clip_mse']=mse;report['clip_psnr_db']=10*math.log10(1/mse)
        stage[0]='metrics-and-artifacts'
        per_frame=[]
        for i,(original,recon) in enumerate(zip(source,reconstructed)):
            frame_mse=float(np.mean((original.astype(np.float64)-recon)**2))
            value=float(structural_similarity(original,recon,data_range=1.,channel_axis=-1,gaussian_weights=True,sigma=1.5,use_sample_covariance=False))
            per_frame.append({'frame':a.start+i,'mse':frame_mse,'psnr_db':10*math.log10(1/frame_mse),'ssim':value})
            shutil.copyfile(files[i],a.output/'original'/files[i].name)
            Image.fromarray(np.rint(recon*255).clip(0,255).astype(np.uint8)).save(a.output/'reconstructed'/files[i].name)
        report['per_frame']=per_frame;report['mean_frame_ssim']=float(np.mean([x['ssim'] for x in per_frame]))
        report['mean_frame_psnr_db']=float(np.mean([x['psnr_db'] for x in per_frame]))
        save_file({'full_clip':latent.cpu().contiguous(),'first_frame_alone':first.cpu().contiguous()},str(a.output/'latents.safetensors'))
        sheet=Image.new('RGB',(1024,3*312),(240,239,232));draw=ImageDraw.Draw(sheet)
        for row,index in enumerate([0,8,16]):
            y=row*312;draw.text((8,y+5),f'Original frame {a.start+index}',fill=(20,20,20));draw.text((520,y+5),'Official Wan VAE reconstruction',fill=(20,20,20))
            sheet.paste(Image.fromarray(images[index]),(0,y+24))
            sheet.paste(Image.open(a.output/'reconstructed'/files[index].name),(512,y+24))
        sheet.save(a.output/'comparison.png')
        report['latents_sha256']=sha(a.output/'latents.safetensors')
        report['comparison_sha256']=sha(a.output/'comparison.png')
        report['status']='passed'
    except Exception as e:
        report['status']='failed';report['error_type']=type(e).__name__;report['error']=str(e)
        raise
    finally:
        stop.set();thread.join(timeout=2);report['peaks_sampled']=peak;report['total_seconds']=time.perf_counter()-started
        save_json(a.output/'metrics.json',report)
    print(json.dumps({k:v for k,v in report.items() if k not in ['input_images','per_frame']},indent=2))

if __name__=='__main__':main()
